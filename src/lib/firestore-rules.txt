// Firestore Security Rules for VietExplore-AI
// Deploy these rules to Firebase Console > Firestore Database > Rules

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Helper functions
    function isAuthenticated() {
      return request.auth != null;
    }
    
    function getUserRole() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role;
    }
    
    function isOwner(resourceData) {
      return request.auth.uid == resourceData.createdBy;
    }
    
    function isAdmin() {
      return getUserRole() == 'admin';
    }
    
    function isModerator() {
      return getUserRole() in ['moderator', 'admin'];
    }
    
    function isContributorOrAbove() {
      return getUserRole() in ['contributor', 'partner', 'moderator', 'admin'];
    }

    // Users collection
    match /users/{userId} {
      allow read: if isAuthenticated();
      allow create: if request.auth.uid == userId && 
                       request.resource.data.role == 'traveler'; // New users start as traveler
      allow update: if request.auth.uid == userId ||
                       isAdmin(); // Users can update themselves, admin can update anyone
      allow delete: if false; // No deletion of users
    }

    // Places collection
    match /places/{placeId} {
      allow read: if resource.data.status == 'published' ||
                     isOwner(resource.data) ||
                     isModerator();
      
      allow create: if isAuthenticated() && 
                       isContributorOrAbove() &&
                       request.resource.data.createdBy == request.auth.uid;
      
      allow update: if isOwner(resource.data) ||
                       isModerator();
      
      allow delete: if isOwner(resource.data) ||
                       isAdmin();
    }

    // Itineraries collection
    match /itineraries/{itineraryId} {
      allow read: if resource.data.isPublic == true ||
                     isOwner(resource.data) ||
                     isModerator();
      
      allow create: if isAuthenticated() &&
                       request.resource.data.createdBy == request.auth.uid;
      
      allow update: if isOwner(resource.data) ||
                       isModerator();
      
      allow delete: if isOwner(resource.data) ||
                       isAdmin();
    }

    // Moderation queue - only moderators and admins
    match /moderation_queue/{itemId} {
      allow read, write: if isModerator();
    }

    // User reports - authenticated users can create, moderators can manage
    match /user_reports/{reportId} {
      allow read: if isOwner(resource.data) ||
                     isModerator();
      
      allow create: if isAuthenticated() &&
                       request.resource.data.reportedBy == request.auth.uid;
      
      allow update: if isModerator();
      allow delete: if isAdmin();
    }

    // Role upgrade requests
    match /role_upgrade_requests/{requestId} {
      allow read: if isOwner(resource.data) ||
                     isAdmin();
      
      allow create: if isAuthenticated() &&
                       request.resource.data.userId == request.auth.uid &&
                       request.resource.data.status == 'pending';
      
      allow update: if isAdmin(); // Only admin can approve/reject
      allow delete: if isAdmin();
    }

    // Analytics events - write only
    match /analytics_events/{eventId} {
      allow read: if isAdmin(); // Only admin can read analytics
      allow create: if true; // Anyone can create events (for tracking)
      allow update, delete: if false; // No updates/deletes
    }
  }
}
